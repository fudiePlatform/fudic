app-counter.fudLo que escribes
@code {
  import { signal } from "@fudic/core";

  const n = signal(0);
  function inc() { n.set(n() + 1); }
}

<head>
  <style>button { border-radius: 999px; }</style>
</head>

<app-counter>
  <template shadowrootmode="open">
    <button @click=@inc>+1</button>
    <output>@n()</output>
  </template>
</app-counter>
