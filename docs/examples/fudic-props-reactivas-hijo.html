prop reactiva de un N3 padre consumida por un hijo — 5 casos según cómo la consume el hijo
El nivel del hijo no lo fija su definición aislada, lo fija cómo consume la prop entrante. Misma regla en todos: el padre inyecta el.prop = ctx.signal antes de anchor.before(el); el setter del hijo discrimina primitivo (rama SSR, sin subscription) vs signal (abre subscription, guarda teardown). Teardown en disconnectedCallback.

Caso 1 — interpolación escalar

N2 · nodo vivo + teardown@prop en textContent + title

app-label.fud

1<span title="@text">@text</span>
2
3@* text es prop entrante. Si llega signal,
4   el nodo se vuelve vivo solo en esa instancia *@

app-label.js emitido

1function renderAppLabel(host, root){
2  let span, unsub=null;
3  const apply=v=>{span.textContent=v;span.setAttribute('title',String(v))};
4  return{
5    create(){span=el('span');apply(host._text??'');root.append(span)},
6    hydrate(){span=root.children[0]},
7    mount(){const p=host._text;
8      if(isSignal(p)){apply(p.peek());unsub=p.subscribe(apply)}
9      else apply(p??'')},
10    update(){}, // fine-grained: la subscription trabaja
11    remove(){if(unsub)unsub()}
12  }
13}

Caso 4 — @if sobre la prop

N3 inducido · update existencial@if (prop) { ... }

app-panel.fud

1@if (show) {
2  <em>presente</em>
3}
4
5@* el @if es su propio render; la prop
6   entrante alimenta su cond via subscription *@

app-panel.js emitido

1function renderIfBlock(ctx, anchor){ // block statement propio
2  let child=null;
3  const mountChild=()=>{child=el('em');child.textContent='presente';anchor.before(child)};
4  return{ create(){if(ctx.cond())mountChild()}, hydrate(){…}, mount(){},
5    update(){ // decisión existencial SOLO del @if
6      const should=ctx.cond();
7      if(should&&!child)mountChild();
8      else if(!should&&child){child.remove();child=null}},
9    remove(){if(child){child.remove();child=null}} };
10}
11function renderAppPanel(host, root){
12  let anchor, ifBlock, unsub=null;
13  return{ create(){…ifBlock=renderIfBlock({cond:()=>truthy(host._show)},anchor);ifBlock.create()},

15      if(isSignal(host._show))unsub=host._show.subscribe(()=>ifBlock.update())}, // estructural

Caso 5 — @foreach sobre prop-array

N3 inducido · reconciliación, estado de instancia@foreach (x of prop) { ... }

app-list.fud

1<ul>
2  @foreach (const item of items) {
3    <li>@item.label</li>
4  }
5</ul>
6
7@* el hijo POSEE la reconciliación.
8   estado de lista = estado de instancia *@

app-list.js emitido

1function renderForeach(ctx, anchor){
2  let rendered=new Map(); // key -> {r, box}  ESTADO DE INSTANCIA
3  const keyOf=i=>i.id;
4  const reconcile=()=>{
5    const next=ctx.items(), nextSet=new Set(next.map(keyOf));
6    for(const[k,e] of[...rendered]) // bajas
7      if(!nextSet.has(k)){e.r.remove();rendered.delete(k)}
8    const m=new Map();
9    for(const item of next){ // altas + reorden + update
10      const k=keyOf(item); let e=rendered.get(k);
11      if(!e){const box={label:item.label};const r=renderItem(box,anchor);r.create();e={r,box}}
12      else{e.box.label=item.label;e.r._node().remove();anchor.before(e.r._node());e.r.update()}
13      m.set(k,e)}
14    rendered=m;
15  };
16  return{ create(){reconcile()}, update(){reconcile()},
17    remove(){for(const{r} of rendered.values())r.remove();rendered.clear()} };
18}
